
CREATE TABLE SP_STORE_BUDGET_HEADS (
    SBH_ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    DEPT_ID NUMBER NOT NULL,
    AY_ID NUMBER NOT NULL,
    HEAD_TITLE VARCHAR2(150) DEFAULT 'Store Items',
    REMARKS VARCHAR2(2000),
    ENTRY_USERNAME VARCHAR2(100),
    ENTRY_DEPT_ID NUMBER,
    CREATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP,
    UPDATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP
);

CREATE TABLE SP_STORE_BUDGET_ITEMS (
    SBI_ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    SBH_ID NUMBER NOT NULL,
    PINNACLE_ITEM_ID VARCHAR2(50),
    ITEM_TITLE VARCHAR2(1000) NOT NULL,
    ITEM_TYPE VARCHAR2(100),
    UNIT VARCHAR2(50),
    APP_UNIT VARCHAR2(50),
    FIN_ACC01_ID VARCHAR2(50),
    FIN_ACC01_TITLE VARCHAR2(500),
    QUANTITY NUMBER NOT NULL,
    RATE NUMBER DEFAULT 0 NOT NULL,
    BUDGET_TYPE VARCHAR2(50),
    PURCHASE_MODE VARCHAR2(50) DEFAULT 'purchase',
    REMARKS VARCHAR2(255),
    AY_ID NUMBER,
    ENTRY_DEPT_ID NUMBER,
    ENTRY_USERNAME VARCHAR2(100),
    CREATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP,
    UPDATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP
);
