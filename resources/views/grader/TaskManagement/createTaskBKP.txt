function addValue2() {
  var textInput = document.getElementById("description");
  var value = textInput.value.trim();

  // Add value to the array and update the list
  if (value !== "") {
    var listItem = document.createElement("li");
    listItem.className = "value-item for_li"; // Add both "value-item" and "for_li" classes
    var serialNumber = document.createElement("span");
    serialNumber.appendChild(document.createTextNode(counter + ") "));
    listItem.appendChild(serialNumber);
    var valueText = document.createElement("span");
    valueText.appendChild(document.createTextNode(value));
    listItem.appendChild(valueText);
    var deleteButton = document.createElement("button");
    deleteButton.className = "delete-button";
    deleteButton.innerHTML = '<i class="fa fa-times"></i>'; // Font Awesome trash icon
    deleteButton.onclick = function () {
      deleteValue(value);
      listItem.remove();
    };
    listItem.appendChild(deleteButton);
    document.getElementById("valueList").appendChild(listItem);
    textInput.value = "";
    
    // Increment the counter for the next value
    counter++;
  }
}

document.getElementById("description").addEventListener("keydown", function(event) {
  if (event.keyCode === 13) { // Check if Enter key was pressed
    event.preventDefault(); // Prevent the default Enter key behavior (new line)
    addValue(); // Call the addValue() function
  }
});

// Function to delete a value from the list
function deleteValue(value) {
  var index = values.indexOf(value);
  if (index !== -1) {
    values.splice(index, 1);
  }
}

// Function to fetch data and append it to the table
function fetchData(selectedStaffId) {
    // Send AJAX request to fetch data
    $.ajax({
        url: '{{ route('fetch_staff_data') }}',
        type: 'POST',
        headers: {
            'X-CSRF-TOKEN': $('meta[name="csrf-token"]').attr('content')
        },
        data: {
            staffId: selectedStaffId
        },
        success: function(response) {
            // Clear existing table rows
            $('#userDataBody').empty();
            $('#user_and_me_div').show();
            $('#type_desc_button').show();
            $('#report_date_time').show();
           // $('#only_button').show();
            $('#show_staff_pannel').show();
            $('#show_staff_pannel').css('width', '100%');
            
            var newRow = '<tr class="check_uncheck_all_tr" style="display:none">' +
                    '<th><input type="checkbox" class="form-control"  name="user_id[]" value="1" onclick="toggleCheckAll(this)"> </th>' +
                    '<th>Check / Uncheck All</th>' +
                    '</tr>';
            $('#userDataBody').append(newRow);  
            newRow = '<tr>';      
            $.each(response.data, function(index, user) {
                // Check if user ID already exists in the list
                var userIdExists = $('input[name="user_id[]"][value="' + user.employee_id + '"]').length > 0;
                if (userIdExists) {
                    return; // Skip this iteration if user ID already exists
                }

                newRow = '<td align="center"><input type="checkbox" checked="checked"  class="form-control"  name="user_id[]" value="' + user.employee_id + '"> </td>' +
                    '<th>' + user.prefix + user.emp_f_name+ ' ' +user.emp_l_name + '</th>';

                $('#userDataBody').append(newRow);
            });
            newRow = '</tr>';
        },
        error: function(xhr, status, error) {
            alert('An error occurred while fetching staff data.');
        }
    });
}
